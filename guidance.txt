function colorClass(c){ return 'b-' + c; }
function dotStyle(c){
  const map = {
    purple:['var(--purple-bg)','var(--purple)','var(--purple-border)'], teal:['var(--teal-bg)','var(--teal)','var(--teal-border)'], coral:['var(--coral-bg)','var(--coral)','var(--coral-border)'], amber:['var(--amber-bg)','var(--amber)','var(--amber-border)'], blue:['var(--blue-bg)','var(--blue)','var(--blue-border)'], green:['var(--green-bg)','var(--green)','var(--green-border)'], pink:['var(--pink-bg)','var(--pink)','var(--pink-border)'], gray:['var(--gray-bg)','var(--gray)','var(--gray-border)']
  }[c];
  return `background:${map[0]};color:${map[1]};border:1px solid ${map[2]}`;
}
function renderNav(){
  const nav = document.getElementById('nav');
  nav.innerHTML = navItems.map(([id,label], i)=>`<button class="${i===0?'active':''}" data-tab="${id}">${label}</button>`).join('');
  nav.addEventListener('click', e => {
    const btn = e.target.closest('button[data-tab]'); if(!btn) return;
    document.querySelectorAll('.nav button').forEach(b=>b.classList.remove('active'));
    btn.classList.add('active');
    document.querySelectorAll('.section').forEach(s=>s.classList.remove('active'));
    document.getElementById(btn.dataset.tab).classList.add('active');
    window.scrollTo({top:0, behavior:'smooth'});
  });
}

function saveChecks(){
  const data = {};
  document.querySelectorAll('input[type="checkbox"]').forEach(cb=>{
    const key = cb.dataset.task || cb.dataset.phase || cb.dataset.quality;
    if(key) data[key]=cb.checked;
  });
  localStorage.setItem('microecon-mvp-progress', JSON.stringify(data));
  updateProgress();
}
function loadChecks(){
  const data = JSON.parse(localStorage.getItem('microecon-mvp-progress') || '{}');
  document.querySelectorAll('input[type="checkbox"]').forEach(cb=>{
    const key = cb.dataset.task || cb.dataset.phase || cb.dataset.quality;
    if(key && data[key]) cb.checked = true;
  });
  updateProgress();
}
function updateProgress(){
  const boxes = [...document.querySelectorAll('.task-check, .quality-check')];
  const done = boxes.filter(b=>b.checked).length;
  const pct = boxes.length ? Math.round(done / boxes.length * 100) : 0;
  document.getElementById('globalProgress').style.width = pct + '%';
  document.getElementById('progressText').textContent = pct + '%';
  phases.forEach(p=>{
    const phaseBox = document.querySelector(`.phase-check[data-phase="${p.id}"]`);
    const taskBoxes = [...document.querySelectorAll(`.task-check[data-task^="${p.id}-"]`)];
    const complete = taskBoxes.length && taskBoxes.every(b=>b.checked);
    if(phaseBox) phaseBox.checked = complete;
    const card = document.getElementById(`phase-${p.id}`); if(card) card.classList.toggle('done', complete);
  });
}
function setupInteractions(){
  document.addEventListener('change', e=>{
    if(e.target.matches('.phase-check')){
      const id = e.target.dataset.phase;
      document.querySelectorAll(`.task-check[data-task^="${id}-"]`).forEach(cb=>cb.checked=e.target.checked);
    }
    if(e.target.matches('input[type="checkbox"]')) saveChecks();
  });
  document.addEventListener('click', e=>{
    const head = e.target.closest('.acc-head');
    if(head) head.closest('.acc-item').classList.toggle('open');
    const filter = e.target.closest('button[data-api-filter]');
    if(filter) renderApi(filter.dataset.apiFilter);
  });
  document.getElementById('expandAll').onclick = () => document.querySelectorAll('.acc-item').forEach(i=>i.classList.add('open'));
  document.getElementById('collapseAll').onclick = () => document.querySelectorAll('.acc-item').forEach(i=>i.classList.remove('open'));
  document.getElementById('resetProgress').onclick = () => { localStorage.removeItem('microecon-mvp-progress'); document.querySelectorAll('input[type="checkbox"]').forEach(cb=>cb.checked=false); updateProgress(); };
  document.getElementById('search').addEventListener('input', e=>{
    const q = e.target.value.trim().toLowerCase();
    document.querySelectorAll('.searchable').forEach(el=>{
      el.classList.toggle('hidden-by-search', q && !el.textContent.toLowerCase().includes(q));
    });
  });
}
